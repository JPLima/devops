# would then age while the runtime's is patched.
#
# This file is for local development and for running the tests:
#
#   python -m venv .venv && . .venv/bin/activate
#   pip install -r lambda/requirements.txt -r tests/requirements.txt
#
# The version tracks what the python3.12 runtime currently ships, so local
# behaviour matches deployed behaviour.
